vector<int> sp_bfs(int s, int n, vector<int> adj[]) {
  queue<int> q;
  q.push(s);
  vector<int> dist(n, INT_MAX);
  dist[s] = 0;
  while (!q.empty()) {
    int x = q.front();
    q.pop();
    for (auto it : adj[x]) {
      int d = dist[x] + 1;
      if (d < dist[it]) {
        dist[it] = d;
        q.push(it);
      }
    }
  }
  return dist;
}

Time O(V + E), space O(V). Vertices that are never reached keep the value INF.
